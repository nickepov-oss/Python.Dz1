import random

heroes = ["Hulk", "Captain America", "Iron-Man", "Spider-Man", "Batman"]

players = {}

while True:
    name = input("Введи своє ім'я (або 'Все' щоб закінчити): ")

    if name.lower() == "все":
        break
    if len(heroes) == 0:
        print("Супергероїв більше немає")
        break
    hero = random.choice(heroes)
    heroes.remove(hero)
    players[name] = hero
print("\nРезультат:")
for name, hero in players.items():
    print(name, "—", hero)
